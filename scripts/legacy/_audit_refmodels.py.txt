def setup_cutoff_multiplicative():
    """Multiplicative cutoff (exp(-E/Ec)) for Band*Cutoff, SBPL*Cutoff."""
    cutoff = Cutoff_powerlaw()
    cutoff.index = 0
    cutoff.index.fix = True
    cutoff.xc = 5e5  # 500 MeV
    cutoff.xc.bounds = (5e4, 1e8)
    cutoff.K = 1
    cutoff.K.fix = True
    cutoff.piv = 1
    cutoff.piv.fix = True
    return cutoff
