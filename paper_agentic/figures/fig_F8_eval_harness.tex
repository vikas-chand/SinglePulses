% F8 — Two harnesses: the runtime harness that walks a burst (left) and the evaluation harness that judges a change
% to it (right). Single-column figure (~3.3 in). COLOUR GRAMMAR (decisions 24/26): blue = code, green = model role,
% white = documents/records, amber = human, dashed = PROPOSED; Source Sans Pro; 1 pt soft-grey round arrows.
% One-row legend (the two-row experiment of r4 collided and bought 0.04 pt of type; the canvas width is set by the two
% 30 mm columns, not the legend — the figure is placed as a full-width float instead, see sec10_discussion.tex).
% NO live count on the artifact (a suite size drifts with any commit: 134 -> 137 within one day when a generator was
% added to the figure-style test). The caption binds the counts to ONE battery record: EVAL_20260903T221534Z.json
% (137 tests collected, 136 passed + 1 xfailed; 25 promoted tables audited, 3 with findings; replay UNBUILT;
% model_id claude-opus-5, git_head 0a506911ad21). Known-results replay = PROPOSED.

\usepackage[T1]{fontenc}
\usepackage[default]{sourcesanspro}

\newcommand{\nohyph}{\hyphenpenalty=10000 \exhyphenpenalty=10000 }
\definecolor{codefill}{RGB}{199,217,239}\definecolor{codeline}{RGB}{59,110,165}
\definecolor{agentfill}{RGB}{226,240,217}\definecolor{agentline}{RGB}{58,125,52}
\definecolor{humanfill}{RGB}{253,236,200}\definecolor{humanline}{RGB}{191,124,0}
\begin{document}
\begin{tikzpicture}[
  font=\sffamily\scriptsize, node distance=2mm and 3mm,
  box/.style={draw=black!35, line width=0.5pt, rounded corners=4pt, fill=white, text width=30mm, align=center, inner sep=2pt, execute at begin node=\nohyph},
  code/.style={box, fill=codefill, draw=codeline, line width=0.8pt},
  agent/.style={box, fill=agentfill, draw=agentline, line width=0.3pt},
  prop/.style={box, dashed, draw=black!45, text=black!70},
  event/.style={box, fill=black!5, draw=black!25},
  human/.style={draw=humanline, thick, diamond, aspect=2.6, fill=humanfill, inner sep=0.5pt, font=\sffamily\scriptsize\bfseries},
  head/.style={font=\sffamily\footnotesize\bfseries, text=black!75},
  arr/.style={-{Stealth[length=2mm,width=1.7mm,round]}, line width=1pt, black!55, line cap=round, line join=round},
  lab/.style={font=\sffamily\scriptsize\itshape, text=black!60, execute at begin node=\nohyph}]

% left column: runtime harness
\node[head] (hl) {runtime harness: walks a burst};
\node[code, below=2.5mm of hl] (prod) {producers: engine, binning, temporal chain, panels, report};
\node[agent, below=of prod] (ver) {fresh-context verifiers: figure, numbers, seed, port; tie reporter; admission gate};
\node[box, below=of ver] (stamp) {stamps, receipts, sidecars: identity, hash, commit};
\node[human, below=3mm of stamp] (gate) {PI gate};
\node[event, below=3mm of gate] (next) {next step; next burst};
\draw[arr] (prod) -- (ver); \draw[arr] (ver) -- (stamp); \draw[arr] (stamp) -- (gate); \draw[arr] (gate) -- (next);
\draw[arr] (next.west) -- ++(-3mm,0) |- (prod.west);

% right column: evaluation harness
\node[head, right=14mm of hl] (hr) {evaluation harness: judges a change};
\node[code, below=2.5mm of hr] (tests) {test suite: lessons-as-tests and schema checks over the products, id guards, compile checks};
\node[code, below=of tests] (audit) {fit-table auditor over every promoted fit table: argmin, rails, tie set, count triple};
\node[prop, below=of audit] (known) {known-results replay of stored solutions and published values (proposed)};
\node[box, below=of known] (rec) {one record per run: PASS / FAIL with the model identity, commit, tree state};
\draw[arr] (tests) -- (audit); \draw[arr] (audit) -- (known); \draw[arr] (known) -- (rec);

% the coupling
\draw[arr] (stamp.east) -- ++(4mm,0) |- node[lab, pos=0.25, right, text width=17mm, align=left] {any change to the runtime, or of the model} (tests.west);
\draw[arr, dashed, -{Bar[width=2.2mm]}] (rec.west) -- ++(-5mm,0) |- (next.east);
\node[lab, below=0.8mm of next.south east, anchor=north, text width=34mm, align=center, xshift=6mm] {a FAIL would block the next step and the freeze (proposed)};

% legend
\node[code, below=13mm of next.south west, anchor=north west, text width=12mm, font=\sffamily\scriptsize] (l1) {blue: code};
\node[agent, right=1.2mm of l1, text width=17mm, font=\sffamily\scriptsize] (l2) {green: model role};
\node[box, right=1.2mm of l2, text width=13mm, font=\sffamily\scriptsize] (l3) {white: records};
\node[event, right=1.2mm of l3, text width=13mm, font=\sffamily\scriptsize] (l3b) {grey: events};
\node[human, right=1.5mm of l3b, aspect=2.2, font=\sffamily\scriptsize\bfseries] (l4) {human};
